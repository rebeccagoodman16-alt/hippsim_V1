Run it in Colab

Quick start

  1. Get the module. Either upload ca1_reduced.py to the Colab session, or open CA1_reduced_validation.ipynb and run its first cell, which writes the file with %%writefile.

  2. Import it and make a parameter set. Params is frozen; change values with replace, so every condition is a new, named object.

    setup
    import numpy as np, matplotlib.pyplot as plt
from dataclasses import replace
import ca1_reduced as m

p  = m.Params()                       # passive defaults (g_h = 0)
ph = replace(p, g_h=2.0)              # Lowet dendritic Ih
print(m.rest_state(ph))               # (V_s, V_d, h) at rest -> (-55.16, -52.19, 0.042)

  3. Run a simulation and plot both compartments. Inputs are functions of the time array; the result is a dict of arrays.

    first run: 8 Hz dendritic drive
    r = m.simulate(ph, T=2000, I_d=m.sine_current(I0=0.0, A=0.05, f_hz=8))

plt.plot(r["t"], r["Vd"], label="dendrite")
plt.plot(r["t"], r["Vs"], label="soma")
plt.xlabel("t (ms)"); plt.ylabel("V (mV)"); plt.legend(); plt.show()

# other keys: r["h"], r["I_ax"] (nA), r["I_h"] (nA), r["spikes"] (ms), r["V0"]

ca1_reduced.py
"""
ca1_reduced.py — Reduced two-compartment CA1 model + validation protocols.

Rung 1 of the ladder: passive / Ih two-compartment (optional LIF soma)
  -> GLIF -> 2-D spiking (e.g. AdEx) -> HH (Golomb soma + dendritic Ih).

Lowet HHca2 is per-area (uF/cm^2, mS/cm^2, uA/cm^2) with C = 1 and equal
compartments; with A = 1e-3 cm^2 its numbers map 1:1 onto nF / uS / nA here.

UNITS (total, not per-area — matches CA1_LIF_Axial_Coupling_EPSP.ipynb)
  t: ms | V: mV | C: nF | g: uS | I: nA
  uS * mV = nA ;  nA / nF = mV/ms
  To use area-normalised literature values: g[uS] = g[mS/cm^2] * A[cm^2] * 1e3,
  C[nF] = c[uF/cm^2] * A[cm^2] * 1e3. Define A_s, A_d before converting.

Sign convention: currents are OUTWARD-positive inside the membrane sum;
I_ax = g_ax * (V_d - V_s) leaves the dendrite and enters the soma.
"""
from __future__ import annotations

from dataclasses import dataclass, replace
import numpy as np


# ----------------------------------------------------------------------------
# Parameters
# ----------------------------------------------------------------------------
@dataclass(frozen=True)
class Params:
    # Passive membrane (total values per compartment)
    C_s: float = 1.0      # nF
    C_d: float = 1.0      # nF
    gL_s: float = 0.05    # uS   -> tau_m = C/gL = 20 ms
    gL_d: float = 0.05    # uS
    E_L: float = -70.0    # mV
    g_ax: float = 0.25    # uS   -> steady Vs/Vd = g_ax/(g_ax+gL_s) = 0.833

    # Dendritic Ih — Lowet et al. 2023 HHca2 (HanLabBU/Lowet-Cell-Reports-2023,
    # biophysical_modeling/ca1.ipynb): h_inf = 1/(1+exp((V+84)/10.2)),
    # tau_h(V) = 1/(exp(-17.9-0.116V) + exp(-1.84+0.09V)) + 0.1 ms,
    # E_h = -32.9 mV, Gh2 = 2 mS/cm^2 (= 2 uS here at A = 1e-3 cm^2).
    # ih_kinetics="simple" restores the old placeholder (constant tau_h).
    ih_kinetics: str = "lowet2023"
    g_h: float = 0.0      # uS   (Lowet default 2.0)
    E_h: float = -32.9    # mV
    Vh_half: float = -84.0  # mV  (h_inf = 0.5)
    k_h: float = 10.2       # mV  (>0: activates with hyperpolarisation)
    tau_h: float = 40.0     # ms  (used only when ih_kinetics == "simple")
    tau_h_scale: float = 1.0  # multiplies tau_h(V); e.g. 1/Q10 per +10 °C (Q10 ~ 3–4 for Ih)
    freeze_h: bool = False  # True: h fixed at its resting value (adds conductance
                            # and shifts rest, but no Ih kinetics) — control for
                            # "is the effect kinetic or just a rest/conductance shift?"

    # Optional somatic subthreshold currents — Golomb et al. 2006 values as used
    # in Lowet 2023 HHca2 (ca1.ipynb). Off by default (g = 0).
    g_M: float = 0.0        # uS   (HHca2 Gm = 1; Ceballos 2025 g_KM = 2.4)
    E_K: float = -90.0      # mV
    theta_z: float = -39.0  # mV   z_inf midpoint
    sigma_z: float = 5.0    # mV
    tau_z: float = 75.0     # ms
    g_NaP: float = 0.0      # uS   (HHca2 Gnap 0–0.41; Ceballos 0.9)
    E_Na: float = 55.0      # mV
    theta_p: float = -47.0  # mV   (Golomb: −47 normal Ca2+, up to −41)
    sigma_p: float = 3.0    # mV

    # Somatic LIF (off by default for subthreshold protocols)
    spiking: bool = False
    V_th: float = -50.0
    V_reset: float = -65.0
    t_ref: float = 2.0    # ms

    # White-noise current amplitude, nA*ms^0.5 (Euler–Maruyama)
    sigma_s: float = 0.0
    sigma_d: float = 0.0

    dt: float = 0.025     # ms


def h_inf(V, p: Params):
    return 1.0 / (1.0 + np.exp((V - p.Vh_half) / p.k_h))


def z_inf(V, p: Params):
    return 1.0 / (1.0 + np.exp(-(V - p.theta_z) / p.sigma_z))


def p_inf(V, p: Params):
    return 1.0 / (1.0 + np.exp(-(V - p.theta_p) / p.sigma_p))


def tau_h(V, p: Params):
    if p.ih_kinetics == "lowet2023":
        return p.tau_h_scale * (1.0 / (np.exp(-17.9 - 0.116 * V) + np.exp(-1.84 + 0.09 * V)) + 0.1)
    return p.tau_h_scale * p.tau_h


def dh_inf_dV(V, p: Params):
    hi = h_inf(V, p)
    return -hi * (1.0 - hi) / p.k_h


# ----------------------------------------------------------------------------
# Exact resting state (Lowet tau_h is 0.5–3 s, so time-settling is unreliable)
# ----------------------------------------------------------------------------
def rest_state(p: Params, I_s0: float = 0.0, I_d0: float = 0.0):
    """Exact steady state (V_s, V_d, h) with h = h_inf(V_d), z = z_inf(V_s)
    and constant currents I_s0, I_d0 (nA)."""
    def Fs(vs, vd):
        return (-p.gL_s * (vs - p.E_L) - p.g_M * z_inf(vs, p) * (vs - p.E_K)
                - p.g_NaP * p_inf(vs, p) * (vs - p.E_Na) + p.g_ax * (vd - vs) + I_s0)

    def vd_of(vs):          # dendrite is monotone in vd for fixed vs -> bisection
        F = lambda vd: (-p.gL_d * (vd - p.E_L) - p.g_h * h_inf(vd, p) * (vd - p.E_h)
                        - p.g_ax * (vd - vs) + I_d0)
        lo, hi = -150.0, 50.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            lo, hi = (mid, hi) if F(mid) > 0 else (lo, mid)
        return 0.5 * (lo + hi)

    if not (p.g_M or p.g_NaP):   # soma linear -> fast 1-D bisection on V_d
        vs_lin = lambda vd: (p.gL_s * p.E_L + p.g_ax * vd + I_s0) / (p.gL_s + p.g_ax)
        F = lambda vd: (-p.gL_d * (vd - p.E_L) - p.g_h * h_inf(vd, p) * (vd - p.E_h)
                        - p.g_ax * (vd - vs_lin(vd)) + I_d0)
        lo, hi = -150.0, 50.0
        for _ in range(100):
            mid = 0.5 * (lo + hi)
            lo, hi = (mid, hi) if F(mid) > 0 else (lo, mid)
        vd = 0.5 * (lo + hi)
        return vs_lin(vd), vd, float(h_inf(vd, p))

    # scan soma voltage for sign changes of the reduced soma equation; take the
    # root closest to E_L (the physiological rest if several exist)
    grid = np.linspace(-200.0, 40.0, 2401)
    G = np.array([Fs(v, vd_of(v)) for v in grid])
    idx = np.where(np.sign(G[:-1]) != np.sign(G[1:]))[0]
    if len(idx) == 0:
        raise ValueError("no resting state found")
    k = idx[np.argmin(abs(grid[idx] - p.E_L))]
    lo, hi = grid[k], grid[k + 1]
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        lo, hi = (mid, hi) if np.sign(Fs(mid, vd_of(mid))) == np.sign(Fs(lo, vd_of(lo))) else (lo, mid)
    vs = 0.5 * (lo + hi); vd = vd_of(vs)
    return vs, vd, float(h_inf(vd, p))


# ----------------------------------------------------------------------------
# Core integrator
# ----------------------------------------------------------------------------
def simulate(p: Params, T: float, I_s=None, I_d=None, g_syn_d=None,
             E_syn: float = 0.0, t_settle: float = 500.0, seed: int | None = None,
             I_hold_s: float = 0.0, I_hold_d: float = 0.0):
    """Forward-Euler (Euler–Maruyama if noisy) integration.

    I_s, I_d, g_syn_d: callables f(t_array) -> array (nA, nA, uS), or None.
    Starts from the exact resting state (rest_state), runs a settle period
    with only the holding currents, then applies I_s/I_d on top of the holds.
    I_hold_s / I_hold_d (nA) are constant for the whole run — use
    match_holding() to set a baseline-matched comparison.
    Returns dict: t, Vs, Vd, h, I_ax, I_h, spikes, V0 = (Vs, Vd) at t = 0.
    """
    dt = p.dt
    n_set, n = int(round(t_settle / dt)), int(round(T / dt))
    t = np.arange(n) * dt
    zeros = np.zeros(n)
    Is = (I_s(t) if I_s else zeros) + I_hold_s
    Id = (I_d(t) if I_d else zeros) + I_hold_d
    gs = g_syn_d(t) if g_syn_d else zeros

    rng = np.random.default_rng(seed)
    sq = np.sqrt(dt)

    def step(vs, vd, h, is_, id_, gsyn, noise=True):
        z = state["z"]
        I_M = p.g_M * z * (vs - p.E_K)
        I_NaP = p.g_NaP * p_inf(vs, p) * (vs - p.E_Na)
        I_ax = p.g_ax * (vd - vs)
        I_h = p.g_h * h * (vd - p.E_h)
        I_syn = gsyn * (vd - E_syn)
        dvd = (-p.gL_d * (vd - p.E_L) - I_h - I_syn - I_ax + id_) / p.C_d
        dvs = (-p.gL_s * (vs - p.E_L) - I_M - I_NaP + I_ax + is_) / p.C_s
        state["z"] = z + dt * (z_inf(vs, p) - z) / p.tau_z
        dh = 0.0 if p.freeze_h else (h_inf(vd, p) - h) / tau_h(vd, p)
        vd_n, vs_n = vd + dt * dvd, vs + dt * dvs
        if noise and (p.sigma_d or p.sigma_s):
            vd_n += p.sigma_d / p.C_d * sq * rng.standard_normal()
            vs_n += p.sigma_s / p.C_s * sq * rng.standard_normal()
        return vs_n, vd_n, h + dt * dh, I_ax, I_h

    # Start at exact rest (with holds), then settle (deterministic, holds only)
    vs, vd, h = rest_state(p, I_hold_s, I_hold_d)
    state = {"z": float(z_inf(vs, p))}
    for _ in range(n_set):
        vs, vd, h, _, _ = step(vs, vd, h, I_hold_s, I_hold_d, 0.0, noise=False)

    Vs, Vd, H, Z = np.empty(n), np.empty(n), np.empty(n), np.empty(n)
    Iax, Ih = np.empty(n), np.empty(n)
    spikes, ref_until = [], -np.inf
    for i in range(n):
        Vs[i], Vd[i], H[i], Z[i] = vs, vd, h, state["z"]
        vs, vd, h, Iax[i], Ih[i] = step(vs, vd, h, Is[i], Id[i], gs[i])
        if p.spiking:
            if t[i] < ref_until:
                vs = p.V_reset
            elif vs >= p.V_th:
                spikes.append(t[i])
                vs, ref_until = p.V_reset, t[i] + p.t_ref

    return dict(t=t, Vs=Vs, Vd=Vd, h=H, z=Z, I_ax=Iax, I_h=Ih,
                spikes=np.array(spikes), V0=(Vs[0], Vd[0]))


# ----------------------------------------------------------------------------
# Input builders
# ----------------------------------------------------------------------------
def step_current(amp, t_on, t_off):
    return lambda t: np.where((t >= t_on) & (t < t_off), amp, 0.0)


def sine_current(I0, A, f_hz, t_on=0.0):
    w = 2 * np.pi * f_hz / 1000.0  # rad/ms
    return lambda t: np.where(t >= t_on, I0 + A * np.sin(w * (t - t_on)), 0.0)


def alpha_like_syn(g_max, t_stim, tau):
    return lambda t: np.where(t >= t_stim, g_max * np.exp(-(t - t_stim) / tau), 0.0)


# ----------------------------------------------------------------------------
# Analytic references (validation targets)
# ----------------------------------------------------------------------------
def analytic_Z(p: Params, f_hz, V0_d: float | None = None):
    """Input (Z_dd) and transfer (Z_sd) impedance, MOhm, dendritic injection.

    Passive if g_h = 0. With g_h > 0 uses the quasi-active linearisation of Ih
    about V0_d (Koch 1984; Hutcheon & Yarom 2000):
      y_d = gL_d + g_h*h0 + i*w*C_d + g_h*(V0-E_h)*h_inf'(V0) / (1 + i*w*tau_h)
    The last term is a slow restorative conductance -> band-pass is possible.
    Purely passive RC compartments can only low-pass (|Z| monotone in f).
    """
    w = 2 * np.pi * np.asarray(f_hz, float) / 1000.0
    y_s = p.gL_s + 1j * w * p.C_s
    y_d = p.gL_d + 1j * w * p.C_d
    if p.g_h:
        V0 = rest_state(p)[1] if V0_d is None else V0_d
        y_d = y_d + p.g_h * h_inf(V0, p)
        if not p.freeze_h:
            y_d = y_d + p.g_h * (V0 - p.E_h) * dh_inf_dV(V0, p) / (1 + 1j * w * tau_h(V0, p))
    g = p.g_ax
    det = (y_d + g) * (y_s + g) - g ** 2
    return (y_s + g) / det, g / det  # uS^-1 = MOhm


def impedance_matrix(p: Params, f_hz, I_hold_s: float = 0.0, I_hold_d: float = 0.0):
    """Full 2x2 quasi-active impedance (MΩ) linearised at the exact rest set by
    the holding currents. Keys: Z_ss (soma in → soma), Z_ds (soma in → dendrite),
    Z_dd (dend in → dend), Z_sd (dend in → soma), plus V0_s, V0_d.
    Includes Ih (dendrite), I_M and I_NaP (soma) when their conductances are > 0.
    I_M adds a delayed restoring term (resonance at depolarised V);
    I_NaP adds an instantaneous NEGATIVE conductance (amplification)."""
    w = 2 * np.pi * np.asarray(f_hz, float) / 1000.0
    vs0, vd0, h0 = rest_state(p, I_hold_s, I_hold_d)
    y_d = p.gL_d + 1j * w * p.C_d
    if p.g_h:
        y_d = y_d + p.g_h * h0
        if not p.freeze_h:
            y_d = y_d + p.g_h * (vd0 - p.E_h) * dh_inf_dV(vd0, p) / (1 + 1j * w * tau_h(vd0, p))
    y_s = p.gL_s + 1j * w * p.C_s
    if p.g_M:
        z0 = z_inf(vs0, p); dz = z0 * (1 - z0) / p.sigma_z
        y_s = y_s + p.g_M * z0 + p.g_M * (vs0 - p.E_K) * dz / (1 + 1j * w * p.tau_z)
    if p.g_NaP:
        p0 = p_inf(vs0, p); dp = p0 * (1 - p0) / p.sigma_p
        y_s = y_s + p.g_NaP * p0 + p.g_NaP * (vs0 - p.E_Na) * dp
    g = p.g_ax
    det = (y_s + g) * (y_d + g) - g ** 2
    return dict(Z_ss=(y_d + g) / det, Z_ds=g / det, Z_dd=(y_s + g) / det, Z_sd=g / det,
                V0_s=vs0, V0_d=vd0)


def match_holding(p: Params, V_target: float, site: str = "d", comp: str = "d"):
    """Constant holding current (nA) at `site` that puts compartment `comp`
    at V_target at rest. Use for baseline-matched comparisons."""
    def V(I):
        vs, vd, _ = rest_state(p, I if site == "s" else 0.0, I if site == "d" else 0.0)
        return vd if comp == "d" else vs
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        lo, hi = (mid, hi) if V(mid) < V_target else (lo, mid)
    return 0.5 * (lo + hi)


def steady_transfer_ratio(p: Params):
    """Passive DC Vs/Vd for dendritic injection."""
    return p.g_ax / (p.g_ax + p.gL_s)

# Protocols
# ----------------------------------------------------------------------------
def lockin(x, t, f_hz):
    """Complex amplitude of x at f (single-bin Fourier projection)."""
    w = 2 * np.pi * f_hz / 1000.0
    return 2 * np.mean((x - x.mean()) * np.exp(-1j * w * t))


def impedance_sweep(p: Params, freqs, A=0.02, I0=0.0, n_cycles=10, t_min=1000.0,
                    site="d", I_hold_s=0.0, I_hold_d=0.0):
    """Simulated impedance (complex, MOhm) from a sinusoid injected at `site`.
    site="d" -> (Z_dd, Z_sd); site="s" -> (Z_ss, Z_ds).
    Discards transients; analyses an integer number of cycles."""
    Zdd, Zsd = [], []
    for f in freqs:
        period = 1000.0 / f
        slow = max(p.C_d / p.gL_d, p.tau_z if p.g_M else 0.0,
                   tau_h(p.E_L, p) if p.g_h else 0.0)
        t_trans = max(t_min, 5 * slow)
        T = t_trans + n_cycles * period
        drive = sine_current(I0, A, f)
        r = simulate(p, T, I_hold_s=I_hold_s, I_hold_d=I_hold_d,
                     **({"I_d": drive} if site == "d" else {"I_s": drive}))
        m = r["t"] >= T - n_cycles * period
        tt = r["t"][m]
        Iin = lockin(I0 + A * np.sin(2 * np.pi * f / 1000 * tt), tt, f)
        here, other = ("Vd", "Vs") if site == "d" else ("Vs", "Vd")
        Zdd.append(lockin(r[here][m], tt, f) / Iin)
        Zsd.append(lockin(r[other][m], tt, f) / Iin)
    return np.array(Zdd), np.array(Zsd)


def resonance_metrics(freqs, Z):
    """f_res and Q = |Z(f_res)| / |Z(f_lowest)|. Q > 1 => band-pass."""
    mag = np.abs(Z)
    k = int(np.argmax(mag))
    return dict(f_res=float(freqs[k]), Q=float(mag[k] / mag[0]))


def sag_ratio(p: Params, amp=-0.1, t_on=200.0, dur=800.0, site="d"):
    """Hyperpolarising step; sag = (V_peak - V_ss) / (V_peak - V_base)."""
    stim = step_current(amp, t_on, t_on + dur)
    r = simulate(p, t_on + dur + 200, **({"I_d": stim} if site == "d" else {"I_s": stim}))
    V = r["Vd"] if site == "d" else r["Vs"]
    win = (r["t"] >= t_on) & (r["t"] < t_on + dur)
    base, peak, ss = V[r["t"] < t_on].mean(), V[win].min(), V[win][-int(50 / p.dt):].mean()
    return dict(sag=(ss - peak) / (base - peak + 1e-12), V_base=base, V_peak=peak, V_ss=ss)


def axial_sweep(p: Params, g_ax_values, amp=0.3, t_on=100.0, t_off=600.0):
    """Dendritic DC step at each g_ax: ΔVd, ΔVs, ratio, steady I_ax."""
    rows = []
    for g in g_ax_values:
        q = replace(p, g_ax=g)
        r = simulate(q, t_off + 100, I_d=step_current(amp, t_on, t_off))
        i_ss = int((t_off - 1) / q.dt)
        dVd, dVs = r["Vd"][i_ss] - r["V0"][1], r["Vs"][i_ss] - r["V0"][0]
        rows.append(dict(g_ax=g, dVd=dVd, dVs=dVs, ratio=dVs / dVd,
                         ratio_analytic=steady_transfer_ratio(q) if not q.g_h else np.nan,
                         I_ax_ss=r["I_ax"][i_ss]))
    return rows


def dt_convergence(p: Params, protocol, dts=(0.1, 0.05, 0.025, 0.0125)):
    """Run protocol(p) -> scalar at several dt; values should converge."""
    return {dt: protocol(replace(p, dt=dt)) for dt in dts}


def noise_trials(p: Params, T=6000.0, n_trials=20, f_hz=8.0, A=0.05, seed0=0):
    """Repeated noisy trials with explicit seeds: SD of Vd, Vs per trial."""
    out = []
    for k in range(n_trials):
        r = simulate(p, T, I_d=sine_current(0.0, A, f_hz), seed=seed0 + k)
        m = r["t"] > 1000
        out.append((r["Vd"][m].std(), r["Vs"][m].std()))
    return np.array(out)  # shape (n_trials, 2)

# Spike-output analysis (ported from Lowet 2023 voltage_imaging scripts)

def classify_bursts(spikes_ms, window_ms=12 / 820 * 1000):
    """figure3_ISI.m rule: a spike is a burst spike if either neighbouring ISI
    < window (12 samples @ 820 Hz ≈ 14.6 ms). Returns bool mask (True = burst)."""
    s = np.asarray(spikes_ms)
    if len(s) < 3:
        return np.zeros(len(s), bool)
    isi = np.diff(s)
    prev = np.r_[np.inf, isi]
    nxt = np.r_[isi, np.inf]
    return (prev < window_ms) | (nxt < window_ms)


def ppc(spikes_ms, f_hz, t_on=0.0):
    """Pairwise phase consistency (spike_field_ppc_adj.m):
    PPC = (N*R^2 - 1)/(N - 1), R = |mean(exp(i*phase))|. Unbiased by N.
    Phase taken relative to the imposed sinusoid (no LFP in the model)."""
    s = np.asarray(spikes_ms)
    N = len(s)
    if N < 10:
        return np.nan, np.nan
    z = np.exp(1j * 2 * np.pi * f_hz / 1000 * (s - t_on)).mean()
    return (N * abs(z) ** 2 - 1) / (N - 1), np.angle(z, deg=True)
